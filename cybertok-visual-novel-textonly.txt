<script>
const stage = document.getElementById('stage');
const dayPill = document.getElementById('dayPill');
const bgmText = document.getElementById('bgmText');

let state = { username: "you", privacy: "", post: "" };

function setDay(d){ dayPill.textContent = d; }
function setBGM(t){ bgmText.textContent = t; }

// render a list of "lines" then a choice block
function render(scene){
  stage.innerHTML = "";
  stage.classList.remove('fade'); void stage.offsetWidth; stage.classList.add('fade');
  if(scene.day) setDay(scene.day);
  if(scene.bgm) setBGM(scene.bgm);

  if(scene.sceneTitle){
    const st = document.createElement('div');
    st.className='scene-title'; st.textContent=scene.sceneTitle;
    stage.appendChild(st);
  }

  const chat = document.createElement('div');
  chat.className='chat';
  (scene.lines||[]).forEach((l,i)=>{
    const m = document.createElement('div');
    m.className = 'msg ' + l.type;
    let text = l.text.replaceAll('{u}', state.username);
    if(l.who){ m.innerHTML = `<span class="who">${l.who}</span>${text}`; }
    else { m.innerHTML = text; }
    m.style.animationDelay = (i*0.06)+'s';
    chat.appendChild(m);
  });
  stage.appendChild(chat);

  if(scene.input){
    const f = document.createElement('div'); f.className='field';
    const inp = document.createElement('input');
    inp.placeholder = scene.input.placeholder; inp.maxLength=24;
    const b = document.createElement('button'); b.className='btn'; b.textContent='Create';
    b.onclick = ()=>{
      let v = inp.value.trim().replace(/^@+/,'');
      if(!v) return;
      state.username = v; scene.input.next();
    };
    inp.addEventListener('keydown',e=>{ if(e.key==='Enter') b.click(); });
    f.appendChild(inp); f.appendChild(b);
    stage.appendChild(f);
    const h=document.createElement('div'); h.className='hint'; h.textContent='@'+'your_handle — pick anything';
    stage.appendChild(h);
    inp.focus();
  }

  if(scene.choices){
    const c = document.createElement('div'); c.className='choices';
    scene.choices.forEach(ch=>{
      const btn = document.createElement('button');
      btn.className = 'choice ' + (ch.kind||'');
      btn.innerHTML = `<span class="tag">${ch.tag}</span><span>${ch.text}</span>`;
      btn.onclick = ch.next;
      c.appendChild(btn);
    });
    stage.appendChild(c);
  }

  if(scene.ending){
    const e = document.createElement('div');
    e.className = 'ending ' + (scene.ending.good?'good':'bad');
    e.innerHTML = `
      <div class="badge">${scene.ending.badge}</div>
      <h2>${scene.ending.title}</h2>
      ${scene.ending.body.map(p=>`<p>${p.replaceAll('{u}',state.username)}</p>`).join('')}
      ${scene.ending.lesson?`<div class="lesson">${scene.ending.lesson}</div>`:''}
      <div class="restart"><button class="btn" onclick="START()">↺ Play Again</button></div>
    `;
    stage.appendChild(e);
  }
  window.scrollTo({top:0,behavior:'smooth'});
}
